+import { useEffect, useRef, useState, useCallback } from 'react';
+import { FleetMap } from '../types/telemetry';
+import { DailyMetricsMap } from '../types/metrics';
+import { getSupabaseClient, isSupabaseConfigured } from '../services/supabaseClient';
+import { calculateDistanceIncrement, CoordinatePoint } from '../services/distanceCalculator';
+
+export function useDailyMetrics(vehicles: FleetMap) {
+  const [dailyMetrics, setDailyMetrics] = useState<DailyMetricsMap>({});
+  const [isLoading, setIsLoading] = useState<boolean>(false);
+
+  // In-memory tracking of previous coordinate per vehicle
+  const prevCoordsRef = useRef<Record<string, CoordinatePoint>>({});
+
+  const todayStr = new Date().toISOString().split('T')[0];
+
+  // Fetch metrics from Supabase if configured
+  const fetchSupabaseMetrics = useCallback(async () => {
+    const supabase = getSupabaseClient();
+    if (!supabase) return;
+
+    try {
+      setIsLoading(true);
+      const { data, error } = await supabase
+        .from('daily_vehicle_metrics')
+        .select('*')
+        .eq('date', todayStr);
+
+      if (!error && data) {
+        setDailyMetrics((prev) => {
+          const next: DailyMetricsMap = { ...prev };
+          for (const row of data) {
+            const vId = row.vehicle_id;
+            const existing = prev[vId];
+            const remoteDistance = Number(row.total_distance_km) || 0;
+            const remoteMaxSpeed = Number(row.max_speed_kmh) || 0;
+
+            next[vId] = {
+              vehicle_id: vId,
+              date: row.date,
+              total_distance_km: existing
+                ? Math.max(existing.total_distance_km, remoteDistance)
+                : remoteDistance,
+              max_speed_kmh: existing
+                ? Math.max(existing.max_speed_kmh, remoteMaxSpeed)
+                : remoteMaxSpeed,
+              avg_speed_kmh: Number(row.avg_speed_kmh) || (existing?.avg_speed_kmh ?? 0),
+              active_minutes: Number(row.active_minutes) || (existing?.active_minutes ?? 0),
+              last_updated_at: row.last_updated_at || existing?.last_updated_at || new Date().toISOString(),
+            };
+          }
+          return next;
+        });
+      }
+    } catch (err) {
+      console.warn('Supabase daily metrics query error:', err);
+    } finally {
+      setIsLoading(false);
+    }
+  }, [todayStr]);
+
+  // Initial fetch and 30-second refetch interval
+  useEffect(() => {
+    if (isSupabaseConfigured()) {
+      fetchSupabaseMetrics();
+      const interval = setInterval(fetchSupabaseMetrics, 30000);
+      return () => clearInterval(interval);
+    }
+  }, [fetchSupabaseMetrics]);
+
+  // Real-time incremental distance accumulation from live telemetry stream
+  useEffect(() => {
+    const updates: Record<string, { distance: number; maxSpeed: number }> = {};
+
+    for (const vehicle of Object.values(vehicles)) {
+      const { vehicle_id, lat, lng, speed, timestamp } = vehicle;
+      const currPoint: CoordinatePoint = { lat, lng, speed, timestamp };
+      const prevPoint = prevCoordsRef.current[vehicle_id];
+
+      if (prevPoint) {
+        const increment = calculateDistanceIncrement(prevPoint, currPoint);
+        if (increment > 0) {
+          updates[vehicle_id] = {
+            distance: increment,
+            maxSpeed: Math.round(speed * 10) / 10,
+          };
+        }
+      }
+
+      prevCoordsRef.current[vehicle_id] = currPoint;
+    }
+
+    if (Object.keys(updates).length > 0) {
+      setDailyMetrics((prev) => {
+        const next: DailyMetricsMap = { ...prev };
+        const nowIso = new Date().toISOString();
+
+        for (const [vehicleId, { distance, maxSpeed }] of Object.entries(updates)) {
+          const existing = prev[vehicleId] || {
+            vehicle_id: vehicleId,
+            date: todayStr,
+            total_distance_km: 0,
+            max_speed_kmh: 0,
+            avg_speed_kmh: 0,
+            active_minutes: 0,
+            last_updated_at: nowIso,
+          };
+
+          const updatedDistance = Math.round((existing.total_distance_km + distance) * 100) / 100;
+          const updatedMaxSpeed = Math.max(existing.max_speed_kmh, maxSpeed);
+
+          next[vehicleId] = {
+            ...existing,
+            total_distance_km: updatedDistance,
+            max_speed_kmh: updatedMaxSpeed,
+            last_updated_at: nowIso,
+          };
+        }
+
+        return next;
+      });
+    }
+  }, [vehicles, todayStr]);
+
+  const totalFleetDistanceKm = Math.round(
+    Object.values(dailyMetrics).reduce((sum, m) => sum + m.total_distance_km, 0) * 100
+  ) / 100;
+
+  return {
+    dailyMetrics,
+    totalFleetDistanceKm,
+    isLoading,
+    refetch: fetchSupabaseMetrics,
+  };
+}

+export const JITTER_THRESHOLD_KM = 0.008; // 8 meters
+export const MIN_SPEED_THRESHOLD_KMH = 1.5; // km/h
+export const MAX_VALID_SPEED_KMH = 150.0; // km/h
+
+const EARTH_RADIUS_KM = 6371.0;
+
+function toRadians(degrees: number): number {
+  return (degrees * Math.PI) / 180.0;
+}
+
+/**
+ * Calculates great-circle distance between two GPS coordinates in kilometers.
+ */
+export function haversineDistance(
+  lat1: number,
+  lng1: number,
+  lat2: number,
+  lng2: number
+): number {
+  const dLat = toRadians(lat2 - lat1);
+  const dLng = toRadians(lng2 - lng1);
+
+  const a =
+    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
+    Math.cos(toRadians(lat1)) *
+      Math.cos(toRadians(lat2)) *
+      Math.sin(dLng / 2) *
+      Math.sin(dLng / 2);
+
+  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
+  return EARTH_RADIUS_KM * c;
+}
+
+export interface CoordinatePoint {
+  lat: number;
+  lng: number;
+  speed: number;
+  timestamp: number;
+}
+
+/**
+ * Computes incremental distance between consecutive telemetry points with noise filtering.
+ */
+export function calculateDistanceIncrement(
+  prev: CoordinatePoint | null | undefined,
+  curr: CoordinatePoint
+): number {
+  if (!prev) return 0;
+
+  const rawDistanceKm = haversineDistance(prev.lat, prev.lng, curr.lat, curr.lng);
+
+  // 1. Filter out parked GPS jitter
+  if (rawDistanceKm < JITTER_THRESHOLD_KM && curr.speed < MIN_SPEED_THRESHOLD_KMH) {
+    return 0;
+  }
+
+  // 2. Reject GPS teleport glitch if implied speed exceeds threshold
+  const timeElapsedHours = Math.abs(curr.timestamp - prev.timestamp) / 3600000;
+  if (timeElapsedHours > 0) {
+    const impliedSpeed = rawDistanceKm / timeElapsedHours;
+    if (impliedSpeed > MAX_VALID_SPEED_KMH) {
+      return 0;
+    }
+  }
+
+  return rawDistanceKm;
+}

+};

+let clientInstance: SupabaseClient | null = null;
+
+export function getSupabaseClient(): SupabaseClient | null {
+  if (!isSupabaseConfigured()) {

+import mqtt from 'mqtt';
+import { createClient } from '@supabase/supabase-js';
+import dotenv from 'dotenv';
+
+dotenv.config();
+
+const BROKER = process.env.MQTT_BROKER || 'mqtt://broker.emqx.io:1883';
+const TOPIC = process.env.MQTT_TOPIC || 'shuttle/fleet/v1/+/telemetry';
+const SUPABASE_URL = process.env.SUPABASE_URL;
+const SUPABASE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY || process.env.SUPABASE_ANON_KEY;
+
+if (!SUPABASE_URL || !SUPABASE_KEY) {
+  console.error('[Worker Error] SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY (or SUPABASE_ANON_KEY) must be provided in tools/.env');
+  console.error('Please create a tools/.env file with:');
+  console.error('  SUPABASE_URL=https://<your-project>.supabase.co');
+  console.error('  SUPABASE_SERVICE_ROLE_KEY=<your-service-role-key>');
+  process.exit(1);
+}
+
+const supabase = createClient(SUPABASE_URL, SUPABASE_KEY, {
+  auth: { persistSession: false },
+});
+
+console.log(`[Worker] Connecting to MQTT broker: ${BROKER}...`);
+const client = mqtt.connect(BROKER, {
+  clientId: `shuttle_ingest_worker_${Math.random().toString(16).slice(2, 8)}`,
+});
+
+// Haversine formula for distance between GPS coordinates in kilometers
+function haversineDistance(lat1, lng1, lat2, lng2) {
+  const toRad = (deg) => (deg * Math.PI) / 180;
+  const dLat = toRad(lat2 - lat1);
+  const dLng = toRad(lng2 - lng1);
+  const a =
+    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
+    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) * Math.sin(dLng / 2);
+  return 6371 * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
+}
+
+// In-memory state tracking per vehicle
+const vehicleState = new Map();
+
+client.on('connect', () => {
+  console.log(`[Worker] Connected to MQTT broker! Subscribing to: ${TOPIC}`);
+  client.subscribe(TOPIC, (err) => {
+    if (err) {
+      console.error('[Worker] Subscribe error:', err.message);
+    } else {
+      console.log('[Worker] Ingestion pipeline is active and listening for telemetry.');
+    }
+  });
+});
+
+client.on('message', async (_topic, message) => {
+  try {
+    const data = JSON.parse(message.toString());
+    const { vehicle_id, lat, lng, speed, timestamp } = data;
+
+    if (!vehicle_id || typeof lat !== 'number' || typeof lng !== 'number') {
+      return;
+    }
+
+    const now = Date.now();
+    const today = new Date().toISOString().split('T')[0];
+    const numericSpeed = typeof speed === 'number' ? speed : 0;
+    const pointTimestamp = typeof timestamp === 'number' ? timestamp : now;
+
+    let state = vehicleState.get(vehicle_id);
+    if (!state) {
+      state = {
+        lastLat: lat,
+        lastLng: lng,
+        lastSpeed: numericSpeed,
+        lastTime: pointTimestamp,
+        pendingDistanceKm: 0,
+        maxSpeedKmh: numericSpeed,
+        currentDate: today,
+        lastFlushTime: now,
+        isFlushing: false,
+      };
+      vehicleState.set(vehicle_id, state);
+    } else {
+      // Midnight rollover for max speed tracking
+      if (state.currentDate !== today) {
+        state.currentDate = today;
+        state.maxSpeedKmh = numericSpeed;
+      } else {
+        state.maxSpeedKmh = Math.max(state.maxSpeedKmh, numericSpeed);
+      }
+
+      const dist = haversineDistance(state.lastLat, state.lastLng, lat, lng);
+
+      // Filter jitter (< 8m when speed < 1.5 km/h) & teleport (> 150 km/h)
+      const elapsedHours = Math.abs(pointTimestamp - state.lastTime) / 3600000;
+      const impliedSpeed = elapsedHours > 0 ? dist / elapsedHours : 0;
+
+      if (!(dist < 0.008 && numericSpeed < 1.5) && impliedSpeed <= 150) {
+        state.pendingDistanceKm += dist;
+      }
+
+      state.lastLat = lat;
+      state.lastLng = lng;
+      state.lastSpeed = numericSpeed;
+      state.lastTime = pointTimestamp;
+    }
+
+    // Flush batch to Supabase every 5 seconds per vehicle (with concurrency lock)
+    if (!state.isFlushing && now - state.lastFlushTime >= 5000 && state.pendingDistanceKm > 0) {
+      state.isFlushing = true;
+      const distanceToFlush = state.pendingDistanceKm;
+      state.pendingDistanceKm = 0;
+      state.lastFlushTime = now;
+
+      try {
+        // Upsert daily metrics
+        const { data: existing, error: fetchErr } = await supabase
+          .from('daily_vehicle_metrics')
+          .select('total_distance_km, max_speed_kmh')
+          .eq('vehicle_id', vehicle_id)
+          .eq('date', today)
+          .maybeSingle();
+
+        if (fetchErr) {
+          console.error(`[Worker Error] Failed to fetch current metrics for ${vehicle_id}:`, fetchErr.message);
+          // Restore uncommitted distance to avoid data loss
+          state.pendingDistanceKm += distanceToFlush;
+          return;
+        }
+
+        const existingDistance = Number(existing?.total_distance_km) || 0;
+        const existingMaxSpeed = Number(existing?.max_speed_kmh) || 0;
+
+        const newTotalDistance = Math.round((existingDistance + distanceToFlush) * 100) / 100;
+        const newMaxSpeed = Math.max(existingMaxSpeed, Math.round(state.maxSpeedKmh * 10) / 10);
+
+        const { error: upsertErr } = await supabase.from('daily_vehicle_metrics').upsert({
+          vehicle_id,
+          date: today,
+          total_distance_km: newTotalDistance,
+          max_speed_kmh: newMaxSpeed,
+          last_updated_at: new Date().toISOString(),
+        });
+
+        if (upsertErr) {
+          console.error(`[Worker Error] Failed to upsert metrics for ${vehicle_id}:`, upsertErr.message);
+          // Restore uncommitted distance on upsert failure
+          state.pendingDistanceKm += distanceToFlush;
+        } else {
+          console.log(`[Worker] Updated ${vehicle_id}: +${distanceToFlush.toFixed(3)} km (Total: ${newTotalDistance} km)`);
+        }
+      } catch (flushErr) {
+        console.error(`[Worker Error] Unexpected error during flush for ${vehicle_id}:`, flushErr.message);
+        state.pendingDistanceKm += distanceToFlush;
+      } finally {
+        state.isFlushing = false;
+      }
+    }
+  } catch (err) {
+    console.error('[Worker] Message processing error:', err.message);
+  }
+});
+
+client.on('error', (err) => {
+  console.error('[Worker] MQTT client error:', err.message);
+});
+
+// Graceful shutdown
+const shutdown = () => {

+  client.end(false, () => {

+    process.exit(0);
+  });

+};
